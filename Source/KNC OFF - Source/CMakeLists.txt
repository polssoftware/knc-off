cmake_minimum_required(VERSION 3.22)
project(KNCOff VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_DIR "" CACHE PATH "JUCE 8.0.12 source checkout")
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
 message(FATAL_ERROR "Set JUCE_DIR to your JUCE source checkout")
endif()
add_subdirectory("${JUCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
juce_add_binary_data(SketchAssets SOURCES Assets/Colon.png Assets/Reference.png Assets/SmakReference.png Assets/Underscore.png Assets/MeterReference.png Assets/DarkReference.png Assets/DarkGlyphs.png Assets/DarkWaveform.png Assets/MenuReference.png Assets/NoteGlyphs.png Assets/SpectrumLines.png Assets/WaveformReference.png Assets/HoverGlyphs.png Assets/Alphabet.png Assets/Sliders.png Assets/Digits.png Assets/Symbols.png)
juce_add_plugin(KNCOff COMPANY_NAME "polS Software" BUNDLE_ID "com.polssoftware.kncoff" PLUGIN_MANUFACTURER_CODE PolS PLUGIN_CODE KnOf FORMATS VST3 PRODUCT_NAME "KNC OFF" VST3_CATEGORIES Fx Dynamics Distortion IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(KNCOff PRIVATE Source/Plugin.cpp Source/Plugin.h Source/Dsp.h)
target_compile_definitions(KNCOff PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(KNCOff PRIVATE SketchAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
option(BUILD_TESTING "Build local verification tools" OFF)
include(CTest)
if(BUILD_TESTING)
 add_executable(EffectsTests Tests/EffectsTests.cpp)
 add_test(NAME EffectFeatures COMMAND EffectsTests)
 add_executable(BankTests Tests/BankTests.cpp)
 add_test(NAME FactoryHeadroom COMMAND BankTests)
 add_executable(DspTests Tests/DspTests.cpp)
 add_test(NAME DSP COMMAND DspTests)
 juce_add_console_app(PluginTests PRODUCT_NAME "KNC OFF verification")
 target_sources(PluginTests PRIVATE Tests/PluginTests.cpp)
 target_link_libraries(PluginTests PRIVATE KNCOff juce::juce_audio_utils juce::juce_dsp)
 add_test(NAME PluginState COMMAND PluginTests)
endif()
